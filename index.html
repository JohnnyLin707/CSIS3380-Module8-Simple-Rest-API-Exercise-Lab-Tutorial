4Schema / Model（MongoDB 教程 Step 6）

题目字段：id: number, unique, required、email: text, required、username: text。MongoDB 教程 Step 6 的 Schema 用了 type/required/trim/min 和 { timestamps: true }，这里照抄格式。Mongoose 没有 text 类型 → 用 String；unique: true 必须自己加。

const mongoose = require('mongoose');

const userSchema = new mongoose.Schema(
  {
    id: { type: Number, required: true, unique: true },
    email: { type: String, required: true, trim: true },
    username: { type: String, trim: true }
  },
  { timestamps: true }
);

module.exports = mongoose.model('User', userSchema);

mongoose.model('User', ...) 会自动使用 users 集合（复数化），符合题目。Mongoose 自带的 _id 不用管，我们用业务字段 id。

5GET 列表 + 单条 + 404（MongoDB Step 8；Build-a-REST Step 6/8）

和老师例子的唯一结构差异：老师用 findById() 查默认 _id；本题要按业务字段 id（数字）查，所以换成 findOne({ id })。Build-a-REST Step 6 特意说明 req.params.id 是字符串要用 Number() 比——这里用 parseId 做转换并校验。
routes/users.js — 第一部分（GET）
const express = require('express');
const User = require('../models/User');

const router = express.Router();

// id in the URL is a string; convert it to a number before querying MongoDB
function parseId(req, res, next) {
  const id = Number(req.params.id);
  if (!Number.isInteger(id)) {
    return res.status(400).json({ message: 'id must be a number.' });
  }
  req.userId = id;
  next();
}

// Send a GET request to /api/users to READ a list of users
router.get('/', async (req, res, next) => {
  try {
    const users = await User.find().sort({ id: 1 });
    res.json(users);
  } catch (err) {
    next(err);
  }
});

// Send a GET request to /api/users/user/:id to READ (view) one user
router.get('/user/:id', parseId, async (req, res, next) => {
  try {
    const user = await User.findOne({ id: req.userId });
    if (user) {
      res.json(user);
    } else {
      res.status(404).json({ message: 'User not found.' });
    }
  } catch (err) {
    next(err);
  }
});
Build-a-REST Step 8 的重点：找不到必须 404 + JSON message，不能返回空 200。这里 if (user) {...} else {...} 的写法和该 PDF 完全一致。
✅ 过关：这一段写完先别急，第 9 节统一测试。

6POST + 校验 + try/catch（MongoDB Step 9；Build-a-REST Step 10/11/12）

老师三步走：MongoDB Step 9 / Build-a-REST Step 10 用 res.status(201).json(...) 创建，express.json() 解析 body；Build-a-REST Step 11 缺字段返回 400；Step 12 用 try/catch（最终 Step 17 改成 next(err)）。错误体在路由里用 { message: ... }（Build-a-REST Step 8/11 的格式），统一错误处理器里用 { error: { message } }（Step 17 的格式）。
routes/users.js — 追加 POST
// Send a POST request to /api/users/newuser to CREATE a new user
router.post('/newuser', async (req, res, next) => {
  try {
    if (req.body.id !== undefined && req.body.email) {
      const user = await User.create({
        id: Number(req.body.id),
        email: req.body.email,
        username: req.body.username
      });
      res.status(201).json(user);
    } else {
      res.status(400).json({ message: 'id and email are required.' });
    }
  } catch (err) {
    next(err);
  }
});
✅ 过关：缺 id 或 email 返回 400，正常创建返回 201。

7PUT + DELETE（MongoDB Step 10/11；Build-a-REST Step 13/14）

两本教材的 PUT 不一样，这里按主教材（MongoDB）走：MongoDB Step 10 明确写 PUT 返回 200 + 更新后的文档，用 { returnDocument: 'after', runValidators: true }（Mongoose 9 写法，旧的 new: true 已废弃）；Build-a-REST Step 13 则用 204。老师原话 “Both are correct REST”，但本题是 MongoDB Lab，所以用 200。DELETE 两本一致，都是 204。
routes/users.js — 追加 PUT / DELETE
// Send a PUT request to /api/users/modify/:id to UPDATE (edit) a user's username
// MongoDB tutorial Step 10: return 200 with the updated document
router.put('/modify/:id', parseId, async (req, res, next) => {
  try {
    if (!req.body.username) {
      return res.status(400).json({ message: 'username is required.' });
    }
    const user = await User.findOneAndUpdate(
      { id: req.userId },
      { username: req.body.username },
      { returnDocument: 'after', runValidators: true }
    );
    if (!user) {
      return res.status(404).json({ message: 'User not found.' });
    }
    res.json(user);
  } catch (err) {
    next(err);
  }
});

// Send a DELETE request to /api/users/delete/:id to DELETE a user
router.delete('/delete/:id', parseId, async (req, res, next) => {
  try {
    const user = await User.findOneAndDelete({ id: req.userId });
    if (!user) {
      return res.status(404).json({ message: 'User not found.' });
    }
    res.status(204).end();
  } catch (err) {
    next(err);
  }
});

module.exports = router;
module.exports.getRandomUser = getRandomUser;
老师 Build-a-REST Step 13 的 ONE REPLY PER REQUEST 原则：每个分支只发一次响应（res.json 或 res.end），否则会报 “Cannot set headers after they are sent”。上面用 return res.status(...).json(...) 提前返回，保证后续 res.json(user) 不会被执行两次。
✅ 过关：PUT 成功 200 且返回更新后的用户名；DELETE 成功 204；找不到都是 404。

8random 用户（Task 5，Build-a-REST Step 15）

题目原文要求的是 GET /api/getrandomuser（注意在 /api 下，不是 /api/users/random）。本教程两个都提供：题目路径 /api/getrandomuser 为主，教材风格的 /api/users/random 为辅，两者共用同一个处理函数。
关于路由顺序：Build-a-REST Step 15 演示的是 /quotes/random 和 /quotes/:id 同一层级时会冲突，固定路径必须写前面。但本题单条路径是 /user/:id（多一层 user），/random 根本不会被它匹配，所以实际不会冲突；我们仍然把 random 写在前面，保持好习惯。
routes/users.js — 把 GET /user/:id 之前那段换成这个
// Send a GET request to /api/users/random to READ (view) a random user
// In the Quote API this had to come before /:id; here the single-user path
// is /user/:id, so there is no clash, but we keep the order as a good habit.
async function getRandomUser(req, res, next) {
  try {
    const count = await User.countDocuments();
    if (count === 0) {
      res.status(404).json({ message: 'No users yet.' });
    } else {
      const skip = Math.floor(Math.random() * count);
      const user = await User.findOne().skip(skip);
      res.json(user);
    }
  } catch (err) {
    next(err);
  }
}
router.get('/random', getRandomUser);

在函数末尾导出它，供 server.js 挂到题目的路径上（见第 7 节代码块最后两行）：module.exports.getRandomUser = getRandomUser;

然后在 server.js 的 app.use('/api/users', userRoutes) 之后加一行：

server.js — 题目要求的路径
// Task 5 path exactly as written in the lab: GET /api/getrandomuser
app.get('/api/getrandomuser', userRoutes.getRandomUser);
✅ 过关：GET /api/getrandomuser 和 GET /api/users/random 都能返回一位随机用户（Task 5 加分）。

9-1 server.js 完整代码（照 Step 17 的两个中间件格式）

server.js — 最终版
require('dotenv').config({ quiet: true });
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');
const userRoutes = require('./routes/users');

const app = express();
const port = process.env.PORT || 3000;

// ------------------------------------------------------------ the plan
// Send a GET request to /api/users to READ a list of users
// Send a GET request to /api/users/user/:id to READ (view) one user
// Send a GET request to /api/getrandomuser to READ (view) one random user
// Send a POST request to /api/users/newuser to CREATE a new user
// Send a PUT request to /api/users/modify/:id to UPDATE (edit) a user's username
// Send a DELETE request to /api/users/delete/:id to DELETE a user

// Middleware: turn a JSON request body into the object req.body
app.use(express.json());
app.use(cors());

// A first JSON route
app.get('/', (req, res) => {
  res.json({ message: 'Welcome to the UserList API' });
});

app.use('/api/users', userRoutes);
app.use('/user', userRoutes);   // the lab wording uses DELETE /user/delete/:id

// Task 5 path exactly as written in the lab: GET /api/getrandomuser
app.get('/api/getrandomuser', userRoutes.getRandomUser);

// ------------------------------------------------ error handling
// 1. No route above matched: make a 404 error and pass it on
app.use((req, res, next) => {
  const err = new Error('Not Found');
  err.status = 404;
  next(err);
});

// 2. The error handler (4 parameters): every next(err) ends up here
app.use((err, req, res, next) => {
  if (err.code === 11000) {
    return res.status(400).json({ error: { message: 'Duplicate id: id must be unique.' } });
  }
  if (err.name === 'ValidationError') {
    return res.status(400).json({ error: { message: err.message } });
  }
  res.status(err.status || 500);
  res.json({
    error: {
      message: err.message
    }
  });
});

// ------------------------------------------------ start
// Connect to MongoDB first; only then start listening for requests
async function start() {
  try {
    await mongoose.connect(process.env.MONGODB_URI);
    console.log('Connected to MongoDB');
    app.listen(port, () => {
      console.log(`UserList API listening on http://localhost:${port}`);
    });
  } catch (err) {
    console.error('Could not connect to MongoDB:', err.message);
    process.exit(1);
  }
}

start();
Step 17 三个要点：① 404 中间件放所有路由之后，制造错误再 next(err)；② 错误处理器必须四个参数 (err, req, res, next)，Express 靠参数个数识别它；③ 路由里的 catch(err){ next(err) } 都会跳到这里，统一返回 JSON，不会出现 Express 默认的 HTML 错误页。
